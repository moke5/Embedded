#### scheduler

- 调度器

```C
#define TS_SCHEDULER_MAX_TASKS 8U

typedef void (*TS_TaskCallback_t)(void);

typedef struct {
    TS_TaskCallback_t callback;    // 任务函数指针
    uint32_t period_ms;  // 任务周期（毫秒）
    uint32_t last_run_ms;    // 上次运行时间（毫秒）
    uint8_t enabled;      // 任务是否启用: 1=启用，0=暂停
} TS_TaskEntry_t;

typedef struct {
    TS_TaskEntry_t tasks[TS_SCHEDULER_MAX_TASKS];
    uint8_t task_count;
} TS_Scheduler_t;
```
